cmake_minimum_required(VERSION 3.16)
project(DualSense4HEN VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(ds4hen STATIC src/hid.c src/ps4_pad.c src/vpad.c platform/ps4/vda.c)
target_include_directories(ds4hen PUBLIC include)
if(MSVC)
  target_compile_options(ds4hen PRIVATE /W4 /WX)
else()
  target_compile_options(ds4hen PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(test_usb_owner tests/test_usb_owner.c src/usb_owner.c)
target_include_directories(test_usb_owner PRIVATE include)
add_test(NAME usb_owner_mock COMMAND test_usb_owner)
add_executable(test_bt_monitor tests/test_bt_monitor.c src/bt_monitor.c src/hid.c)
target_include_directories(test_bt_monitor PRIVATE include)
add_test(NAME bt_monitor COMMAND test_bt_monitor)
set_tests_properties(bt_monitor PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(test_transport_research tests/test_transport_research.c src/acl_input.c src/socket_gate.c src/hid.c)
target_include_directories(test_transport_research PRIVATE include)
if(NOT MSVC)
  target_compile_options(test_transport_research PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME transport_research COMMAND test_transport_research)
set_tests_properties(transport_research PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE ds4hen)
add_test(NAME core COMMAND test_core)
set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT MSVC)
  add_executable(test_hidcontrol_audit tests/test_hidcontrol_audit.c platform/ps4/probe.c)
  target_include_directories(test_hidcontrol_audit PRIVATE include)
  target_compile_definitions(test_hidcontrol_audit PRIVATE D4H_HID_CONTROL_ABI_DUMP)
  target_compile_options(test_hidcontrol_audit PRIVATE -Wall -Wextra -Werror)
  add_test(NAME hidcontrol_audit_mock COMMAND test_hidcontrol_audit)
  foreach(variant IN ITEMS registered passive connected)
    add_executable(test_bt_${variant} tests/test_bt.c platform/ps4/bt_observer.c)
    target_include_directories(test_bt_${variant} PRIVATE include)
    target_compile_options(test_bt_${variant} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(variant STREQUAL "passive")
      target_compile_definitions(test_bt_${variant} PRIVATE D4H_BT_PASSIVE)
    endif()
    if(variant STREQUAL "connected")
      target_compile_definitions(test_bt_${variant} PRIVATE D4H_BT_PASSIVE D4H_BT_CONNECTED)
      target_sources(test_bt_${variant} PRIVATE src/bt_monitor.c)
    endif()
    add_test(NAME bt_${variant}_mock COMMAND test_bt_${variant})
  endforeach()
endif()
set(D4H_PAD_HEADER "" CACHE FILEPATH "Optional OpenOrbis orbis/_types/pad.h for Pad snapshot mocks")
if(D4H_PAD_HEADER)
  configure_file("${D4H_PAD_HEADER}" "${CMAKE_BINARY_DIR}/test-headers/orbis/_types/pad.h" COPYONLY)
  add_executable(test_pad_snapshot tests/test_pad_snapshot.c platform/ps4/pad_snapshot.c)
  target_include_directories(test_pad_snapshot PRIVATE include)
  target_include_directories(test_pad_snapshot SYSTEM PRIVATE "${CMAKE_BINARY_DIR}/test-headers")
  add_test(NAME pad_snapshot_mock COMMAND test_pad_snapshot)
endif()
set(D4H_USB_HEADER "" CACHE FILEPATH "Optional PS4 SDK libusb.h for USB mock tests")
if(D4H_USB_HEADER AND NOT MSVC)
  configure_file("${D4H_USB_HEADER}" "${CMAKE_BINARY_DIR}/test-headers/libusb.h" COPYONLY)
  add_executable(test_usb tests/test_usb.c platform/ps4/usb_probe.c src/hid.c src/ps4_pad.c)
  target_include_directories(test_usb PRIVATE include)
  target_include_directories(test_usb SYSTEM PRIVATE "${CMAKE_BINARY_DIR}/test-headers")
  target_compile_options(test_usb PRIVATE -Wall -Wextra -Wpedantic -Werror -include "${CMAKE_CURRENT_SOURCE_DIR}/tests/usb_compat.h")
  add_test(NAME usb_mock COMMAND test_usb)
endif()
